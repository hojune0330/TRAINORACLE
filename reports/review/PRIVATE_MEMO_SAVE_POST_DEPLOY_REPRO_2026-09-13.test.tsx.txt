import React from "react"
import { cleanup, fireEvent, render, screen, waitFor } from "@testing-library/react"
import { afterEach, beforeEach, expect, it } from "vitest"
import { RaceForm } from "./RaceForm"
import { PrivateMemoSaveSetup } from "./PrivateMemoSaveSetup"
import { deleteEntry, loadEntriesWithPrivateMemos, restoreDeletedEntry, savePrivateEntry } from "../../domain/journal-store"
import { createRecoveryCode } from "../../domain/account/private-note-crypto"
import { clearSessionRecoveryCode, loadSessionRecoveryCode, saveSessionRecoveryCode } from "../../domain/account/private-note-sync"
import { setActiveLocalAccount } from "../../domain/account/local-journal-ownership"

beforeEach(() => { localStorage.clear(); sessionStorage.clear(); setActiveLocalAccount(null); clearSessionRecoveryCode() })
afterEach(() => { cleanup(); localStorage.clear(); sessionStorage.clear(); setActiveLocalAccount(null); clearSessionRecoveryCode() })

const entry = (id: string) => ({ id, kind: "race", date: "2026-09-13", savedAt: "2026-09-13T00:00:00.000Z", syncState: "local", stage: "pre", record: "", rank: "", result: "", memo: `synthetic-${id}`, memoPurpose: "PRIVATE_SELF_ONLY" })

it("REPRO: ready UI offers no recovery input after the session code disappears", async () => {
  saveSessionRecoveryCode(createRecoveryCode())
  render(<RaceForm targetDate="2026-09-13" onDone={() => {}} />)
  fireEvent.click(screen.getByRole("radio", { name: "나만의 메모" }))
  fireEvent.change(screen.getByLabelText("경기 메모"), { target: { value: "synthetic-review" } })
  expect(screen.getByText(/암호화 준비가 됐어요/)).toBeVisible()
  clearSessionRecoveryCode()
  fireEvent.click(screen.getByRole("button", { name: /^저장/ }))
  expect(await screen.findByRole("alert")).toHaveTextContent("복구 코드를 준비")
  expect(screen.getByText(/암호화 준비가 됐어요/)).toBeVisible()
  expect(screen.queryByLabelText("기존 복구 코드")).toBeNull()
  expect(screen.queryByRole("button", { name: /코드 만들기/ })).toBeNull()
})

it("REPRO: trash-only old ciphertext permits a new key and restoration makes both keys fail preparation", async () => {
  const oldCode = createRecoveryCode()
  saveSessionRecoveryCode(oldCode)
  expect((await savePrivateEntry(entry("old"))).ok).toBe(true)
  expect(deleteEntry("old")).toMatchObject({ ok: true, trashed: true })
  clearSessionRecoveryCode()
  render(<PrivateMemoSaveSetup onReady={() => {}} />)
  fireEvent.click(screen.getByRole("button", { name: /처음 사용해요/ }))
  expect(screen.getByLabelText("보관할 복구 코드")).toBeVisible()
  fireEvent.click(screen.getByRole("checkbox", { name: "복구 코드를 따로 보관했어요" }))
  fireEvent.click(screen.getByRole("button", { name: "이 코드로 저장 준비" }))
  await screen.findByText(/암호화 준비가 됐어요/)
  const newCode = loadSessionRecoveryCode()!
  expect(newCode).not.toBe(oldCode)
  expect((await savePrivateEntry(entry("new"))).ok).toBe(true)
  expect(restoreDeletedEntry("old").ok).toBe(true)
  cleanup()
  clearSessionRecoveryCode()
  render(<PrivateMemoSaveSetup onReady={() => {}} />)
  for (const candidate of [oldCode, newCode]) {
    fireEvent.change(screen.getByLabelText("기존 복구 코드"), { target: { value: candidate } })
    fireEvent.click(screen.getByRole("button", { name: "이 코드로 저장 준비" }))
    await waitFor(() => expect(screen.getByRole("alert")).toHaveTextContent("맞지 않거나"))
    expect(loadSessionRecoveryCode()).toBeNull()
  }
  // Ciphertexts still exist: the defect is mixed-key recovery, not proven destruction.
  saveSessionRecoveryCode(oldCode)
  expect((await loadEntriesWithPrivateMemos()).some(value => value.kind === "race" && value.memo === "synthetic-old")).toBe(true)
  saveSessionRecoveryCode(newCode)
  expect((await loadEntriesWithPrivateMemos()).some(value => value.kind === "race" && value.memo === "synthetic-new")).toBe(true)
})
